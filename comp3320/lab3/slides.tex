\section{Intro}

\begin{frame}{What we will do}
  \begin{enumerate}\setlength\itemsep{8pt}
    \item The compiler \textbf{auto-vectorises} simple loops. \cplx{(week 4, lecture 1)}
    \item The \textbf{Vector Class Library} lets you write SIMD by hand,
      without intrinsics. \cplx{(week 4, lecture 2)}
    \item The naive matrix multiply runs at \textbf{1--2\% of peak}.
      \cplx{(week 5, lecture 1)}
  \end{enumerate}
  \bigskip
  \begin{block}{The whole lab is one problem}
    $C = A \times B$, square, doubles - the lecture's kernels, verbatim.
    The harness times any variant, prints
    $\text{GFLOP/s} = 2n^{3}/\text{walltime}$, and compares
    \textbf{every element} of the answer.
  \end{block}
  \medskip
  Questions 1--7 fit the session; 8 is an extension.
  \textbf{Question 5 is the centrepiece}
\end{frame}

  \end{cookiecode}

\begin{frame}[fragile]{Pin every timing run}
  \begin{cookiecode}[title={define pin() once, put it in front of every timing command}]{bash}
pin() {
    local cpu=$(awk '/Cpus_allowed_list/{split($2,a,",");
                     split(a[1],b,"-");print b[1]}' /proc/self/status)
    numactl --localalloc --physcpubind=$cpu "$@"
}
  \end{cookiecode}
  \medskip
  \begin{itemize}\setlength\itemsep{4pt}
    \item Unpinned, these measurements vary by \textbf{3$\times$} run to run.
    \item Even pinned, run everything more than once - two numbers inside
      your own scatter are \emph{the same number}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Q1: the wrong node is a \emph{silent} failure}
  \texttt{-march=native} decides whether a \texttt{Vec8d} is one real 512-bit
  register or an emulation out of two \texttt{Vec4d}. Both \textbf{build and
  run and give the same answer} - only the numbers stop meaning anything.
  \begin{cookiecode}[title={check when you land}]{bash}
lscpu | grep -oE 'avx512[a-z0-9_]*' | sort -u   # prints nothing = wrong node
./instrset_probe                                # Q1: INSTRSET = ?
  \end{cookiecode}
  \pause
  \begin{itemize}\setlength\itemsep{4pt}
    \item \texttt{INSTRSET} is the one number VCL derives from the compiler
      flags - look yours up in \texttt{vectorclass/instrset.h}.
    \item \texttt{sizeof(Vec8d)} is 64 bytes \emph{either way}: the type does
      not change on a narrower machine, only the instructions under it.
  \end{itemize}
\end{frame}

\section{What the compiler does on its own}

\begin{frame}[fragile]{Q2 - two kernels, one poll: which inner loop vectorises?}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
      \begin{cookiecode}[title={naive: inner loop is a reduction}]{C}
for (i...)
  for (j...) {
    double sum = 0.0;
    for (k...)
      sum += A[i*n+k] * B[k*n+j];
    C[i*n+j] = sum;
  }
      \end{cookiecode}
    \column{.48\textwidth}
      \begin{cookiecode}[title={flip-k: inner loop is n independent updates}]{C}
memset(C, 0, ...);
for (k...)
  for (i...)
    for (j...)
      C[i*n+j] +=
        A[i*n+k] * B[k*n+j];
      \end{cookiecode}
  \end{columns}
  \medskip
  Use \texttt{make vecreport} to verify.
\end{frame}

\begin{frame}{Q2, the reveal: both - and that is not the interesting part}
  \begin{itemize}\setlength\itemsep{6pt}
    \item gcc 15 vectorises even the reduction, \emph{in order} - it has
      learned to keep your summation order and vectorise anyway.
    \item The tell is the \textbf{widths}. The report says ``16 byte
      vectors'', ``32 byte vectors'' - and the lecture's decoder is
      16 = SSE, 32 = AVX, \textbf{64 = AVX-512}.
  \end{itemize}
  \bigskip
  \begin{block}{This machine has 512-bit registers. Nothing got 64 bytes.}
    Hold that thought until question 6.
  \end{block}
\end{frame}

\begin{frame}[fragile]{Q3 - what loop order alone is worth}
  \begin{cookiecode}[title={measure both}]{bash}
pin ./exe_mm 1024 0 3      # naive
pin ./exe_mm 1024 1 3      # flip-k
  \end{cookiecode}
  \medskip
  \begin{center}
    \begin{tabular}{@{}lcc@{}}
      \toprule
      & naive & flip-k \\
      \midrule
      GFLOP/s & ? & ? \\
      \bottomrule
    \end{tabular}
  \end{center}
  \medskip
  \pause
  The factor should startle you: \textbf{nobody wrote a line of SIMD.}
  Explain it with the lecture's stride analysis - which array does the naive
  inner loop read with \textbf{stride n}, and what does that cost per
  iteration against a stride-1 walk?
\end{frame}

\begin{frame}[fragile]{Q4 - the answer moved and nobody asked it to}
  The harness says flip-k is \textbf{not bit-identical} to naive -
  differences around the 13th digit. There is no \texttt{-ffast-math}
  anywhere in this lab, and both loops add each $C_{ij}$'s $k$ terms in the
  same order. Where did it come from?
  \pause
  \begin{cookiecode}[title={ask the assembly}]{bash}
make asm
awk '/^mm_naive:/,/\t.size/'  matmul.s | grep -cE 'vfmadd'
awk '/^mm_flip_k:/,/\t.size/' matmul.s | grep -cE 'vfmadd'
  \end{cookiecode}
  \begin{block}{Fused multiply-add rounds once, not twice}
    One kernel's multiply-adds were fused into FMA, the other's mostly were
    not - and gcc contracts \textbf{by default} (\texttt{-ffp-contract=fast}).
    Neither answer is wrong; the fused one is the \emph{more} accurate. If the
    last bit matters to you, you have to say so.
  \end{block}
\end{frame}

\section{Write it yourself}

\begin{frame}{Q5 - the centrepiece: \texttt{mm\_vcl}}
  Write flip-k again, doing \textbf{eight j's at once} with \texttt{Vec8d}.
  Everything you need is listed in \texttt{matmul.cpp}:
  \medskip
  \begin{itemize}\setlength\itemsep{5pt}
    \item broadcast one A element: \texttt{Vec8d a(A[i*n+k]);}
    \item \texttt{load} 8 of B and C, \texttt{mul\_add(a, b, c)}, \texttt{store}
    \item the tail, when 8 does not divide n:
      \texttt{load\_partial} / \texttt{store\_partial}
      \cplx{(lecture 4-2, ``advice 3'')}
  \end{itemize}
  \bigskip
  \begin{block}{The bar is bit-identical vs \texttt{mm\_flip\_k} - at every n, including 1003}
    Same fused operations in the same order; vectorising \emph{across} j
    reorders nobody's sum. The harness pre-fills C with NaN, so forgetting to
    zero it fails loudly.
  \end{block}
  \medskip
  \cplx{Write it yourself. It is the only SIMD you produce in this course, the
  harness knows if it is right, and the exam room has no AI in it.}
\end{frame}

\begin{frame}[fragile]{Q6 - so who vectorised it better, you or gcc?}
  \begin{cookiecode}[title={look at the registers each of you used}]{bash}
objdump -d exe_mm | awk '/<mm_flip_k>:/,/ret/' | grep -oE '%[xyz]mm' | sort -u
objdump -d exe_mm | awk '/<mm_vcl>:/,/ret/'    | grep -oE '%[xyz]mm' | sort -u
  \end{cookiecode}
  \pause
  You are in \texttt{zmm} because you \emph{named the type}; the compiler
  chose \texttt{ymm} unless told:
  \begin{cookiecode}[title={the lecture's extra flag}]{bash}
make exe_mm_wide           # -mprefer-vector-width=512
pin ./exe_mm_wide 1024 1 3
  \end{cookiecode}
  Compare all three: on a loop this simple the honest result is a
  \textbf{near tie} - so what did hand-writing buy, and what did it cost?
\end{frame}

\begin{frame}{Q7 - how far up the mountain are you?}
  Work out this core's peak, then place yourself on it:
  \[
    \text{peak} \approx
    \underbrace{2}_{\text{FMA units}} \times
    \underbrace{8}_{\text{doubles}} \times
    \underbrace{2}_{\text{FLOP/FMA}} \times
    \text{clock} \;\approx\; 100\ \text{GFLOP/s on one core}
  \]
  \pause
  \begin{itemize}\setlength\itemsep{5pt}
    \item Your best kernel is a \textbf{single-digit percentage} of that -
      and the gap is not vectorisation; you are already 8 wide.
    \item It is the \textbf{memory hierarchy}: B gets re-fetched endlessly.
      Blocking (Q8) is the first bite; a tuned BLAS gets \textbf{$\sim$90\% of
      peak}, and \texttt{OpenBLAS} is a \texttt{module load} away.
    \item The lecture's last word stands: \emph{you do not write your own
      matmul in production}. The other 95\% of the node is \textbf{threads} -
      that is week 8.
  \end{itemize}
\end{frame}

\section{Blocking and beyond}

\begin{frame}[fragile]{Q8 (extension)}
  \begin{cookiecode}[title={sweep the block size}]{bash}
for bs in 4 8 16 32 64; do pin ./exe_mm 1024 3 3 $bs; done
  \end{cookiecode}
  \medskip
  \begin{itemize}\setlength\itemsep{5pt}
    \item Here it lands \textbf{between} naive and flip-k - not on top, as
      the lecture's plot had it. \texttt{make vecreport} shows why: blocking
      fixed cache reuse but kept the stride-n reads.
      \cplx{Analyse $\to$ iterate $\to$ verify - the lecture's own method.}
    \item The check line: blocked is bit-identical to \emph{neither}
      reference. You reassociated the sum yourself - no compiler flag
      required to move an answer.
    \item ``Can we still do better?'' - read algorithmica's matmul chapter
      for the rest of the ladder, and note where \textbf{call BLAS} sits.
  \end{itemize}
\end{frame}

\section{Narrower numbers}

\begin{frame}[fragile]{Q9 - the other way to get more lanes}
  Half the bits, twice the lanes. Same kernel, three edits:
  \texttt{Vec8d}$\to$\texttt{Vec16f}, \texttt{double}$\to$\texttt{float},
  step 8$\to$16.
  \begin{cookiecode}[title={the same multiply in four precisions, same inputs}]{bash}
pin ./exe_prec 1024 3
  \end{cookiecode}
  \medskip
  \begin{itemize}\setlength\itemsep{4pt}
    \item This is the one place in the lab where you should expect the
      \textbf{textbook 2x} and actually get it.
    \item Now read the \emph{other} column. \texttt{exe\_prec} prints
      $\|C-\text{ref}\|/\|\text{ref}\|$ against a double reference: how many
      digits did that speed cost you?
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Q10 (optional) - ``just use FP16''}
  The last two rows are half precision, both \textbf{provided}, differing in
  exactly one thing: the type the sum is accumulated into.
  \pause
  \begin{itemize}\setlength\itemsep{4pt}
    \item That gap is the entire cost of rounding every partial sum back to
      \textbf{10 mantissa bits}, 1024 times per output element. One of those
      two rows is shippable; it is the one tensor cores implement.
    \item And the speed is the surprise:
  \end{itemize}
  \begin{cookiecode}[title={check the hardware before you believe a timing}]{bash}
lscpu | grep -o avx512_fp16 || echo "no native FP16 arithmetic on this node"
  \end{cookiecode}
  \cplx{AVX-512-FP16 is Sapphire Rapids and later; Cascade Lake predates it,
  so gcc keeps binary16 semantics by converting to float and back around
  every operation - scalar. Both half rows come out \emph{slower than
  double}.}
  \medskip
  \textbf{Split your answer}: which half is a fact about this hardware, and
  which is a fact about the format?
\end{frame}
