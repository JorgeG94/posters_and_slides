\begin{frame}{ANUGA}{CDAC (India) \textperiodcentered\ ANU \textperiodcentered\ GA \textperiodcentered\ NCI --- via an MOU}
  \begin{columns}[T,onlytextwidth]
    \column{.53\textwidth}

        \item Faster code $\Rightarrow$ better, more accurate simulations in less time.
        \item Python-orchestrated, \textbf{C backend} for speed.
      \end{itemize}
    \column{.43\textwidth}
      \begin{block}{State of the code when I started}
        \begin{itemize}
          \item 4 parallel backends: serial C, SIMD C, OpenMP, OpenACC
          \item GPU: sparse, poorly integrated, untested
          \item Great unit \& automated testing
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{ANUGA: towards a GPU-accelerated codebase}
  \begin{itemize}\setlength\itemsep{6pt}
    \item For GPUs to be worth it, \textbf{everything} needs to be accelerated.
    \item You are only as fast as your slowest bit. (Amdahl's law)
    \item Re-architect how the codebase works.
  \end{itemize}
\end{frame}

\begin{frame}{How does GPU acceleration work?}
  \begin{itemize}
    \item GPUs are fast because they have \textbf{thousands} of cores; CPUs have hundreds.
    \item The problem must be very parallel to suit a GPU \cplx{(sometimes all you need is a large problem)}.
    \item Problems are \textbf{memory-} or \textbf{compute-bound}:
      \begin{itemize}
        \item \emph{Memory:} how fast you can access the elements for a computation.
        \item \emph{Compute:} how fast you can do the arithmetic on them.
      \end{itemize}
    \item Separate memory address spaces --- the CPU has its memory, the GPU its own.
      \begin{itemize}
        \item How do they talk? The \textbf{bus} --- a limiting factor you must optimize around.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Writing very fast GPU code: A six step program}
  \begin{enumerate}\setlength\itemsep{4pt}
    \item Allocate all memory at the start and reuse it.
    \item Minimize large CPU$\leftrightarrow$GPU transfers during the main computational loop.
    \item Make sure the problem is large enough to use the GPU efficiently.
    \item Expose the maximum amount of parallelism possible
    \item Leave only very small, insignificant operations on the CPU.
    \item Profit.
  \end{enumerate}
\end{frame}

\begin{frame}{Expressing parallelism}
  \begin{itemize}
    \item Loops are inherently \textbf{serial}: \texttt{do i=1, i=2, \dots}
    \item Parallelism is expressed via \textbf{pragmas}: \texttt{!\$omp parallel do}.
    \item \emph{You} tell the program a region can run in parallel --- get it wrong and the code is wrong.
  \end{itemize}
  \begin{block}{Lessons in parallelism}
    \begin{itemize}
      \item No side effects outside the loop (no writing to global state; thread safety).
      \item Minimize branching (\texttt{if}, \texttt{switch}, \texttt{case}).
      \item Make it easy for the compiler to optimize --- simple is always best.
      \item No fancy object-oriented shenanigans inside hot loops.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}[fragile]{ANUGA's hot loop: fluxes over triangles}
  \begin{columns}[T,onlytextwidth]
    \column{.56\textwidth}
      \begin{cookiecode}[title={compute\_fluxes --- the main kernel}]{C}
// one pragma flips CPU -> GPU
#pragma omp target teams loop
for (n = 0; n < n_triangles; n++) {
  for (i = 0; i < 3; i++) {
    // local Riemann solve -> edge flux
    flux_edge(n, i, ...);
  }
  explicit_update[n] = ...;       // writes only n
}
      \end{cookiecode}
    \column{.42\textwidth}
      \begin{itemize}
        \item One loop over \textbf{every triangle} --- \texttt{do n = 1, n\_triangles}.
        \item Each triangle is \textbf{independent}: reads its neighbours, writes only itself.
        \item Exactly the ``no side effects'' contract $\Rightarrow$ safe to parallelize.
        \item Most of the runtime lives here --- accelerate this loop and you accelerate ANUGA.
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Road to GPU ANUGA}{MPI vs OpenMP --- and why you want both}
  \begin{itemize}
    \item ANUGA was already parallelized with \textbf{MPI} (distributed memory).
    \item ANUGA is \textbf{memory-bound} due to unstructured grids.
  \end{itemize}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
      \begin{block}{MPI}
        Divides the domain into subdomains, each executed concurrently --- reduces \textbf{memory pressure} by making each problem smaller.
      \end{block}
    \column{.48\textwidth}
      \begin{block}{OpenMP}
        Keeps the full domain, but launches \textbf{threads} to evaluate it in parallel.
      \end{block}
  \end{columns}
  \vspace{.3em}
  \begin{exampleblock}{The mix that gets you GPUs}
    One MPI subdomain \textbf{per GPU}; each GPU gets a very large subdomain.
  \end{exampleblock}
\end{frame}

\begin{frame}[fragile]{How to get GPUs?}
  \begin{columns}[T,onlytextwidth]
    \column{.5\textwidth}
      \begin{itemize}
        \item Because ANUGA was already OpenMP-parallel, the trick was to switch\\
          \texttt{\#pragma omp parallel for} $\to$\\
          \texttt{\#pragma omp target teams loop}.
        \item The real pain was the \textbf{memory}:
          \begin{itemize}
            \item Refactor the backend from a single global domain state into an \emph{owning} object.
            \item Allocate all memory upfront and manage it efficiently.
          \end{itemize}
      \end{itemize}
    \column{.48\textwidth}
      \begin{cookiecode}[title={the evolve loop}]{Python}
domain.set_memory(n_triangles, quantities)
domain.distribute(n_gpus)
for t in range(t_end):
    domain.evolve(t)
    if t == t_print:
        domain.print_console_update()
      \end{cookiecode}
  \end{columns}
\end{frame}

\begin{frame}{What did we achieve?}{ANUGA on GPUs}
  \begin{itemize}\setlength\itemsep{5pt}
    \item CDAC simulation: \textbf{4 hours on 4416 cores} $\to$ the same sim on \textbf{4 nodes (16 GPUs) in under an hour}.
    \item $\sim$\textbf{10$\times$} savings in SUs \cplx{(speedup is trickier to compare across different units)}.
    \item A realistic $\sim$16M-triangle simulation used to need a big compute node --- now it runs on your \textbf{laptop} \cplx{(given a GPU with $\geq$\,8\,GB of memory)}.
  \end{itemize}
\end{frame}

\section{Standard parallelism}

\begin{frame}[fragile]{\texttt{do concurrent}: the next best thing since sliced bread}
  \begin{columns}[T,onlytextwidth]
    \column{.5\textwidth}
      \begin{cookiecode}[title=Standard Fortran]{Fortran}
do concurrent (k=1:nk,j=1:nj,i=1:ni)
  c(i,j,k) = a(i,j,k)
end do
      \end{cookiecode}
    \column{.5\textwidth}
      \begin{cookiecode}[title=Directive equivalents]{Fortran}
!$omp target teams loop collapse(3)
do k = 1, nk
  do j = 1, nj
    do i = 1, ni
      c(i,j,k) = a(i,j,k)
    end do
  end do
end do
      \end{cookiecode}
  \end{columns}
  \vspace{.2em}
  {\footnotesize
  \begin{itemize}\setlength\itemsep{1pt}
    \item A \textbf{contract}: every iteration is independent and thread-safe; for offload, called procedures must be \texttt{pure}.
    \item Idiomatic, no vendor pragmas --- and a routine containing \texttt{do concurrent} can itself stay \texttt{pure}.
  \end{itemize}}
\end{frame}

  \centering

    \item Minimize transfers: work backwards from the kernels, keep data on the GPU between steps.
    \item OpenMP mapping is portable across NVIDIA, AMD and Intel.
  \end{itemize}
\end{frame}

      \end{itemize}

      \cplx{Work $\Theta(N)$ \textperiodcentered\ depth $\Theta(n_z)$ \textperiodcentered\ columns fully parallel}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Motif 4 --- Iterative column solve}
  \begin{cookiecode}[title=data-dependent iteration count]{Fortran}
do concurrent (j, i)             ! columns parallel
  iter = 0
  do while (.not. converged .and. iter < max)
    ! update kappa(k) from shear; solve tridiagonal column; check residual
    iter = iter + 1
  end do
end do
  \end{cookiecode}
  \begin{itemize}
    \item Shear-driven mixing (kappa-shear), implicit / convective adjustment.
    \item \cplx{Work $\Theta(N\cdot it)$ \textperiodcentered\ depth $\Theta(n_z\cdot it)$}
    \item GPU pain: \textbf{thread divergence} --- columns converge at different rates.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Motif 5 --- Barotropic subcycle (2-D split-explicit)}
  \begin{cookiecode}[title={fast surface mode, small steps}]{Fortran}
do m = 1, n_bt                   ! n_bt ~ 10-60, dt_bt = dt/n_bt
  eta = eta - dt_bt * div(D*U)   ! 2-D continuity   (pointwise + halo)
  U   = U   - dt_bt * (g*grad(eta) + ...)  ! 2-D momentum (stencil)
  call halo_exchange(eta, U)     ! every substep
end do
  \end{cookiecode}
  \begin{itemize}
    \item Cheap per substep but \textbf{sequential} in $m$. \cplx{Work $\Theta(H\,n_{bt})$ \textperiodcentered\ depth $\Theta(n_{bt})$}
    \item Tiny 2-D kernels $\times\,n_{bt}$ launches $\times$ halo each $\Rightarrow$ \textbf{openacc overhead bound}.
  \end{itemize}
\end{frame}

\begin{frame}{Motif 6 --- Halo exchange: pack / transfer / unpack}
  \centering
  \begin{tikzpicture}[node distance=12mm,
      ph/.style={cookie node,align=center,minimum height=1.15cm,minimum width=2.2cm,font=\scriptsize},
      lbl/.style={font=\scriptsize\bfseries,text=cookieAccent,align=center}]
    \node[ph] (faces) {tile-edge faces\\\footnotesize(strided)};
    \node[ph,right=of faces] (buf1) {send buffer\\\footnotesize(contiguous)};
    \node[ph,right=of buf1] (buf2) {recv buffer\\\footnotesize neighbour};
    \node[ph,right=of buf2] (ghost) {ghost cells};
    \draw[cookie arrow] (faces) -- node[above,lbl]{pack\\\emph{gather}} (buf1);
    \draw[cookie arrow] (buf1) -- node[above,lbl]{transfer\\\emph{MPI}} (buf2);
    \draw[cookie arrow] (buf2) -- node[above,lbl]{unpack\\\emph{scatter}} (ghost);
  \end{tikzpicture}
  \vspace{.5em}
  \begin{itemize}
    \item \textbf{pack / unpack} are on-device \emph{gather / scatter} kernels --- depth $\Theta(1)$, \textbf{non-coalesced}, work $\propto$ surface.
    \item \textbf{transfer} is the MPI comm. \cplx{$\Theta(\text{halo}\cdot(n_i{+}n_j)\cdot n_z/\sqrt{P})$ vs compute $\Theta(N/P)$ --- strong scaling set by surface/volume.}
    \item Every variable $\times$ direction $\times$ phase is its own kernel $\Rightarrow$ \textbf{many tiny kernels}, so the exchange is \textbf{OpenACC launch-overhead bound}, not bandwidth bound.
    \item \textbf{GPUDirect:} buffers stay on the device --- no host staging.
  \end{itemize}
\end{frame}
